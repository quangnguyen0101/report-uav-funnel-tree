\section{Danh mục viết tắt}

\begin{description}
  \item[LiDAR] (Light Detection and Ranging): Cảm biến đo khoảng cách bằng laser
  \item[RRT] (Rapidly-exploring Random Tree): Cây ngẫu nhiên khám phá nhanh
  \item[RRT*] (Asymptotically Optimal Rapidly-exploring Random Tree): Cây
        ngẫu nhiên khám phá nhanh tối ưu tiệm cận
  \item[PRM] (Probabilistic Roadmap): Bản đồ đường đi xác suất
  \item[SLAM] (Simultaneous Localization and Mapping): Định vị và dựng bản
        đồ đồng thời
\end{description}
